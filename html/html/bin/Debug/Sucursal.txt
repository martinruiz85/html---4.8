GO
SET QUOTED_IDENTIFIER ON
GO
IF NOT EXISTS (SELECT * FROM sys.triggers WHERE object_id = OBJECT_ID(N'[dbo].[LogSucursal_Update]'))
EXEC dbo.sp_executesql @statement = N'CREATE TRIGGER LogSucursal_Update
ON Sucursal
AFTER INSERT, UPDATE
AS
BEGIN
    SET NOCOUNT ON;

	-- REGISTRO DE NUEVOS INGRESOS	
    INSERT INTO HistoricoEstatusSucursal(SucursalId,EstatusId,NuevoEstatusId,FechaRegistro)
    SELECT i.Id,
	       null,
		   i.EstatusId,
		   GETDATE()
      FROM inserted i
      LEFT JOIN deleted d ON i.ID = d.ID
     WHERE d.ID IS NULL;

    -- VERIFICAR QUE SOLO SE HAYA MODIFICADO LA COLUMNA ''ESTATUS''
    IF (UPDATE(EstatusId)) 
    BEGIN
        IF EXISTS (SELECT 1 
                         FROM inserted i
						INNER JOIN deleted d ON i.Id = d.Id
                        WHERE (i.EstatusId != d.EstatusId)) BEGIN

            INSERT INTO HistoricoEstatusSucursal(SucursalId,EstatusId,NuevoEstatusId,FechaRegistro)
            SELECT i.Id,
				   d.EstatusId,
			       i.EstatusId,
				   GETDATE()
              FROM inserted i
			 INNER JOIN deleted d ON i.Id = d.Id
             WHERE (i.EstatusId != d.EstatusId)
        END
    END
END;

' 
GO
ALTER TABLE [dbo].[Sucursal] ENABLE TRIGGER [LogSucursal_Update]
GO

 if UPDATE(LastUser)

 end
 else
 begin
 insert into LogCambioDeEstatus
 select @IdSucursal,@RecordDate, @EstatusId, @User	
 end
end
