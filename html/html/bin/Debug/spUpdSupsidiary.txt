CREATE proc [dbo].[spUpdSupsidiary]	
@Id INT,    
@Nombre  NVARCHAR(40),        
@BaseDeDatos NVARCHAR(510),       
@Email NVARCHAR(500),  
@Id_Cliente INT,         
@status BIT,   
@usuarioLogeado VARCHAR(50),  
@TipoSucursalId INT,  
@EstatusId INT,
@Longitud Varchar(30),
@Latitud Varchar(30)
    
AS      
DECLARE @Date DATETIME      
BEGIN     
    
UPDATE sucursal     
SET nombre = @Nombre,     
basededatos = @BaseDeDatos,     
email = @Email,  
[status] = @status,  
Id_Cliente = @Id_Cliente,  
TipoSucursalId = @TipoSucursalId,  
EstatusId = @EstatusId,
Latitud = @Latitud,
Longitud = @Longitud
WHERE id = @Id     
  
SELECT @Date = GETDATE()    
EXECUTE WansoftAdmin..spInsSupportWebLog @Id, 4,2,@Date, @usuarioLogeado                      
END

GO
